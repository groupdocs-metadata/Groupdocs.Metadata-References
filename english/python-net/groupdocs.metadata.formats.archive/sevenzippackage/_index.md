### Example

```python
from groupdocs.metadata import Metadata, SevenZipRootPackage, Constants

encoding = 'cp866'  # encoding 866

with Metadata(Constants.InputSevenZip) as metadata:
    root = metadata.get_root_package(SevenZipRootPackage)

    print(root.seven_zip_package.total_entries)

    for file in root.seven_zip_package.files:
        print(file.name)
        print(file.compressed_size)
        print(file.modification_date_time)
        print(file.uncompressed_size)
        print(file.raw_name.decode(encoding))
```
