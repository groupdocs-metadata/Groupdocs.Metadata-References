### Example

```python
from groupdocs.metadata import Metadata, WavRootPackage

with Metadata(Constants.InputWav) as metadata:
    root = metadata.get_root_package(WavRootPackage)
    if root.wav_package is not None:
        print(root.wav_package.audio_format)
        print(root.wav_package.bits_per_sample)
        print(root.wav_package.block_align)
        print(root.wav_package.byte_rate)
        print(root.wav_package.number_of_channels)
        print(root.wav_package.sample_rate)
```
