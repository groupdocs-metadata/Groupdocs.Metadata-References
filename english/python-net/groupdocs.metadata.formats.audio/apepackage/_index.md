### Example

```python
from groupdocs.metadata import Metadata, MP3RootPackage

with Metadata(Constants.MP3WithApe) as metadata:
    root = metadata.get_root_package(MP3RootPackage)

    if root.ApeV2:
        print(root.ApeV2.Album)
        print(root.ApeV2.Title)
        print(root.ApeV2.Artist)
        print(root.ApeV2.Composer)
        print(root.ApeV2.Copyright)
        print(root.ApeV2.Genre)
        print(root.ApeV2.Language)
```
