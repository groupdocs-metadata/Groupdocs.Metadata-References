### Example

```python
from groupdocs.metadata import Metadata, Constants, VCardRootPackage

def print_array(values):
    if values is not None:
        for value in values:
            print(value)

with Metadata(Constants.InputVcf) as metadata:
    root = metadata.get_root_package(VCardRootPackage)

    for vcard in root.vcard_package.cards:
        # Print most preferred work phone numbers and work emails
        filtered = vcard.filter_work_tags().filter_preferred()
        print_array(filtered.communication_recordset.telephones)
        print_array(filtered.communication_recordset.emails)
```
